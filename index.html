Clean up old profile tables if they exist DROP TABLE IF EXISTS player_profile_socials; DROP TABLE IF EXISTS player_profile_contacts; DROP TABLE IF EXISTS player_profile_tournaments; DROP TABLE IF EXISTS player_profile; -- 2. Main Player Profile Attributes (Academic, Athletic, Personal Info) CREATE TABLE IF NOT EXISTS player_profile ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, category TEXT NOT NULL, -- 'Personal', 'Academics', 'Athletic' label TEXT NOT NULL, -- e.g. 'Cell Phone', 'NCAA ID', '2026 High School Stats' value TEXT, -- The actual value to copy display_order INT DEFAULT 0, created_at TIMESTAMPTZ DEFAULT NOW() ); -- 3. Social Media Accounts (Twitter/X, Instagram, Facebook) CREATE TABLE IF NOT EXISTS player_profile_socials ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, platform TEXT NOT NULL, -- 'Twitter / X', 'Instagram', 'Facebook' handle TEXT NOT NULL, -- e.g. '@FrankieW2028' display_order INT DEFAULT 0, created_at TIMESTAMPTZ DEFAULT NOW() ); -- 4. Key Contacts (High School Coach, Club Coach, Counselor) CREATE TABLE IF NOT EXISTS player_profile_contacts ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, role_title TEXT NOT NULL, -- 'High School Coach', 'Club Team & Coach', 'High School Counselor' contact_name TEXT NOT NULL, -- e.g. 'Nina Green' email TEXT, -- e.g. 'agreen@fifeschools.com' phone TEXT, -- e.g. '253-260-9798' display_order INT DEFAULT 0, created_at TIMESTAMPTZ DEFAULT NOW() ); -- 5. Tournaments with start/end dates and locations CREATE TABLE IF NOT EXISTS player_profile_tournaments ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, tournament_name TEXT NOT NULL, start_date DATE, end_date DATE, city TEXT, state TEXT, display_order INT DEFAULT 0, created_at TIMESTAMPTZ DEFAULT NOW() );
